<Note>
This page focuses on optimal configuration and benchmark results for DeepSeek-V4-Flash on the Ascend NPU. For environment
setup, model weight download, feature configuration, and deployment instructions, etc., see the
[DeepSeek-V4-Flash Model Tutorial](/docs/hardware-platforms/ascend-npus/model-deployment/tutorials/deepseek_v4_flash).

On the A3 Series, each card has 2 dies, so `--tp-size` is twice the card count. On the 950PR/DT Series, each card is a
single die, so the card count, the device count, and `--tp-size` match. See
[Ascend NPU Reference](/docs/hardware-platforms/ascend-npus/reference/glossary#hardware) for details.
</Note>

### DeepSeek-V4-Flash W8A8 1P1D 16P IN8K OUT1K 50ms

**Model**: DeepSeek-V4-Flash

**Hardware**: A3 Series Products

**Cards**: 16

**Deploy Mode**: PD Disaggregation

**Quantization**: W8A8 INT8

**Dataset**: 8k+1k

**TPOT**: 50ms

#### Model Deployment

The prefill instance uses TP16 with DP attention of width 8 and prefill context parallelism, which extends the
supported input length. The decode instance uses TP16 with DP attention of width 16 and DSpark speculative decoding.
For short inputs, remove `--enable-prefill-cp --cp-strategy interleave` from the prefill command.

```bash Command
# ============================================================
# Before running, update the following variables:
#   P_IP: prefill node IP address
#   D_IP: decode node IP address
#   ASCEND_MF_STORE_URL: prefill node IP with port
#   MODEL_PATH: path to the model weights directory
#   HCCL_SOCKET_IFNAME: network interface name for HCCL
#   GLOO_SOCKET_IFNAME: network interface name for Gloo
# ============================================================


echo performance | tee /sys/devices/system/cpu/cpu*/cpufreq/scaling_governor
sysctl -w vm.swappiness=0
sysctl -w kernel.numa_balancing=0
sysctl -w kernel.sched_migration_cost_ns=50000

unset https_proxy
unset http_proxy
unset HTTPS_PROXY
unset HTTP_PROXY
unset ASCEND_LAUNCH_BLOCKING

source /usr/local/Ascend/ascend-toolkit/set_env.sh
source /usr/local/Ascend/nnal/atb/set_env.sh
source /usr/local/Ascend/ascend-toolkit/latest/opp/vendors/customize/bin/set_env.bash
source /usr/local/Ascend/ascend-toolkit/latest/opp/vendors/custom_transformer/bin/set_env.bash

export DEEP_NORMAL_MODE_USE_INT8_QUANT=1
export FORCE_DRAFT_MODEL_NON_QUANT=1
export HCCL_OP_EXPANSION_MODE=AIV
export INF_NAN_MODE_FORCE_DISABLE=1
export PYTORCH_NPU_ALLOC_CONF=expandable_segments:True
export SGLANG_DSV4_FP4_EXPERTS=False
export SGLANG_ENABLE_OVERLAP_PLAN_STREAM=1
export SGLANG_ENABLE_SPEC_V2=1
export SGLANG_ENABLE_TP_MEMORY_INBALANCE_CHECK=0
export SGLANG_OPT_BF16_FP32_GEMM_ALGO=torch
export SGLANG_OPT_DEEPGEMM_HC_PRENORM=False
export SGLANG_OPT_FP8_WO_A_GEMM=0
export SGLANG_OPT_FUSE_WQA_WKV=0
export SGLANG_OPT_USE_FUSED_HASH_TOPK=False
export SGLANG_OPT_USE_OVERLAP_STORE_CACHE=False
export SGLANG_OPT_USE_TILELANG_MHC_POST=False
export SGLANG_OPT_USE_TILELANG_MHC_PRE=False
export SGLANG_RAGGED_VERIFY_MODE=static
export SGLANG_SET_CPU_AFFINITY=1
export STREAMS_PER_DEVICE=32

P_IP=('<your prefill ip>')
D_IP=('<your decode ip>')

export ASCEND_MF_STORE_URL="tcp://<your prefill ip>:24669"
export SGLANG_DISAGGREGATION_BOOTSTRAP_TIMEOUT=1800
export SGLANG_DISAGGREGATION_WAITING_TIMEOUT=1800
# Prefill runs with context parallelism while decode does not. This switch lets
# the decode workers accept the CP layout used by prefill.
export SGLANG_DISAGGREGATION_ALL_CP_RANKS_TRANSFER=1

MODEL_PATH=/path/to/model-weights

LOCAL_HOST1=`hostname -I|awk -F " " '{print$1}'`
LOCAL_HOST2=`hostname -I|awk -F " " '{print$2}'`
echo "${LOCAL_HOST1}"
echo "${LOCAL_HOST2}"

# prefill
for i in "${!P_IP[@]}";
do
    if [[ "$LOCAL_HOST1" == "${P_IP[$i]}" || "$LOCAL_HOST2" == "${P_IP[$i]}" ]];
    then
        echo "Prefill -> ${P_IP[$i]}"
        export DEEPEP_HCCL_BUFFSIZE=2600
        export DEEPEP_NORMAL_LONG_SEQ_PER_ROUND_TOKENS=8192
        export DEEPEP_NORMAL_LONG_SEQ_ROUND=8
        export DEEPEP_NORMAL_COMBINE_ENABLE_LONG_SEQ=1
        export GLOO_SOCKET_IFNAME=<network-interface>
        export HCCL_SOCKET_IFNAME=<network-interface>
        export SGLANG_DEEPEP_NUM_MAX_DISPATCH_TOKENS_PER_RANK=128
        export SGLANG_DISAGG_PREFILL_EARLY_SEND_CACHED_PREFIX=1
        export SGLANG_ENABLE_WAR_BARRIER=1
        export SGLANG_FORCE_COARSE_WAR_BARRIER=1

        python3 -m sglang.launch_server \
            --model-path ${MODEL_PATH} \
            --host ${P_IP[$i]} \
            --port 8000 \
            --page-size 128 \
            --tp-size 16 \
            --trust-remote-code \
            --device npu \
            --attention-backend dsv4 \
            --watchdog-timeout 9000 \
            --disaggregation-mode prefill \
            --disaggregation-transfer-backend ascend \
            --disaggregation-bootstrap-port 8998 \
            --mem-fraction-static 0.62 \
            --prefill-max-requests 256 \
            --max-prefill-tokens 70000 \
            --chunked-prefill-size 131072 \
            --max-running-requests 256 \
            --dp-size 8 \
            --enable-dp-attention \
            --tokenizer-worker-num 16 \
            --moe-a2a-backend deepep \
            --deepep-mode normal \
            --quantization modelslim \
            --enable-dp-lm-head \
            --kv-cache-dtype auto \
            --load-balance-method round_robin \
            --enable-prefill-cp \
            --cp-strategy interleave \
            --disable-cuda-graph
        break
    fi
done

# decode
for i in "${!D_IP[@]}";
do
    if [[ "$LOCAL_HOST1" == "${D_IP[$i]}" || "$LOCAL_HOST2" == "${D_IP[$i]}" ]];
    then
        echo "Decode -> ${D_IP[$i]}"
        export DEEPEP_HCCL_BUFFSIZE=1200
        export DEEPEP_NORMAL_LONG_SEQ_ROUND=8
        export DEEPEP_NORMAL_LONG_SEQ_PER_ROUND_TOKENS=2048
        export DEEPEP_NORMAL_COMBINE_ENABLE_LONG_SEQ=1
        export GLOO_SOCKET_IFNAME=<network-interface>
        export HCCL_BUFFSIZE=1200
        export HCCL_SOCKET_IFNAME=<network-interface>
        export SGLANG_DEEPEP_NUM_MAX_DISPATCH_TOKENS_PER_RANK=128
        export SGLANG_DISAGG_PREFILL_EARLY_SEND_CACHED_PREFIX=1
        export SGLANG_NPU_SPLIT_SHARED_EXPERT_OVERLAP=1
        export SGLANG_NPU_USE_MULTI_STREAM=1

        python3 -m sglang.launch_server \
            --model-path ${MODEL_PATH} \
            --host ${D_IP[$i]} \
            --port 8001 \
            --page-size 128 \
            --tp-size 16 \
            --trust-remote-code \
            --device npu \
            --attention-backend dsv4 \
            --watchdog-timeout 9000 \
            --disaggregation-mode decode \
            --disaggregation-transfer-backend ascend \
            --mem-fraction-static 0.71 \
            --max-running-requests 256 \
            --dp-size 16 \
            --enable-dp-attention \
            --moe-a2a-backend deepep \
            --deepep-mode low_latency \
            --quantization modelslim \
            --enable-dp-lm-head \
            --kv-cache-dtype auto \
            --load-balance-method round_robin \
            --cuda-graph-bs-decode 1 2 3 4 5 6 \
            --speculative-algorithm DSPARK \
            --speculative-draft-model-path "${MODEL_PATH}" \
            --speculative-draft-model-quantization modelslim \
            --speculative-draft-attention-backend ascend \
            --speculative-num-draft-tokens 6
        break
    fi
done
```

```bash Command
# ============================================================
# Before running, replace the following placeholders:
#   <your prefill ip>: prefill node IP address
#   <your decode ip>: decode node IP address
# ============================================================

python -m sglang_router.launch_router \
    --pd-disaggregation \
    --prefill http://<your prefill ip>:8000 8998 \
    --decode http://<your decode ip>:8001 \
    --host 127.0.0.1 \
    --port 6688 \
    --policy cache_aware
```

### DeepSeek-V4-Flash FP8 1P1D 8P PD Disaggregation

**Model**: DeepSeek-V4-Flash

**Hardware**: 950PR/DT Series

**Cards**: 8

**Deploy Mode**: PD Disaggregation

**Quantization**: FP8

**Dataset**: 8k+1k

**TPOT**: -

#### Model Deployment

Use the official block FP8 checkpoint. Both instances run on one 950PR/DT Series node with TP8 and DP attention of
width 4. Prefill and decode communicate through MemFabric-Hybrid over `device_urma`, so source the MemFabric-Hybrid
environment on both nodes. Decode runs DSpark speculative decoding with the same parameter block as the A3 Series.

```bash Command

#   P_IP: prefill node IP address
#   D_IP: decode node IP address
#   ASCEND_MF_STORE_URL: prefill node IP with port
#   MODEL_PATH: path to the model weights directory
#   HCCL_SOCKET_IFNAME: network interface name for HCCL
#   GLOO_SOCKET_IFNAME: network interface name for Gloo
# ============================================================

sysctl -w kernel.numa_balancing=0

unset https_proxy
unset http_proxy

source /usr/local/Ascend/ascend-toolkit/latest/opp/vendors/customize/bin/set_env.bash
source /usr/local/Ascend/ascend-toolkit/latest/opp/vendors/custom_transformer/bin/set_env.bash
source /usr/local/Ascend/ascend-toolkit/latest/set_env.sh
source /usr/local/Ascend/nnal/atb/set_env.sh
source /usr/local/memfabric_hybrid/set_env.sh

export ASCEND_USE_FIA=1
export ASCEND_MF_TRANSFER_PROTOCOL="device_urma"
export HCCL_HOST_SOCKET_PORT_RANGE=auto
export MF_HYBM_USE_VMM_SEGMENT=1

export SGLANG_DSV4_FP4_EXPERTS=True
export SGLANG_DSPARK_FAST_KERNEL=0
export SGLANG_ENABLE_OVERLAP_PLAN_STREAM=1
export SGLANG_ENABLE_SPEC_V2=1
export SGLANG_OPT_BF16_FP32_GEMM_ALGO=torch
export SGLANG_OPT_DEEPGEMM_HC_PRENORM=False
export SGLANG_OPT_FP8_WO_A_GEMM=False
export SGLANG_OPT_FUSE_WQA_WKV=0
export SGLANG_OPT_USE_FUSED_HASH_TOPK=False
export SGLANG_OPT_USE_OVERLAP_STORE_CACHE=False
export SGLANG_OPT_USE_TILELANG_MHC_POST=False

echo "${LOCAL_HOST1}"
echo "${LOCAL_HOST2}"

# prefill
for i in "${!P_IP[@]}";
do

    then
        echo "Prefill -> ${P_IP[$i]}"
        export DEEPEP_HCCL_BUFFSIZE=2600
        export DEEPEP_NORMAL_COMBINE_ENABLE_LONG_SEQ=1
        export DEEPEP_NORMAL_LONG_SEQ_PER_ROUND_TOKENS=2048
        export DEEPEP_NORMAL_LONG_SEQ_ROUND=8
        export GLOO_SOCKET_IFNAME=<network-interface>
        export HCCL_SOCKET_IFNAME=<network-interface>
        export SGLANG_ENABLE_WAR_BARRIER=1
        export SGLANG_FORCE_COARSE_WAR_BARRIER=1

        python3 -m sglang.launch_server \
            --model-path ${MODEL_PATH} \
            --host ${P_IP[$i]} \
            --port 8000 \
            --tp-size 8 \
            --dp-size 4 \
            --enable-dp-attention \
            --enable-dp-lm-head \
            --trust-remote-code \
            --device npu \
            --attention-backend ascend \
            --watchdog-timeout 9000 \
            --disaggregation-mode prefill \
            --disaggregation-transfer-backend ascend \
            --disaggregation-bootstrap-port 8998 \
            --tokenizer-worker-num 16 \
            --mem-fraction-static 0.79 \
            --max-prefill-tokens 70000 \
            --chunked-prefill-size 65536 \
            --max-running-requests 64 \
            --quantization fp8 \
            --kv-cache-dtype auto \
            --moe-a2a-backend deepep \
```
